Clients

tsdb has clients for C, Rust, Python and Java. Each one talks to tsdbd, and each one can also run the database inside your program, with no server. Pick your language once; every page remembers it.

Every program on this page is one our tests build and run, or a README section they run or compile. A check in make test makes sure the page still shows exactly that code.

1. Get the client

First build tsdb and start a server, in the top directory of the checkout. The server lets in only clients that log in, so first add a user, admin, who may do everything (a "superuser"). Its password goes in a file only you can read; every example on this page logs in with it:

make
mkdir -p /tmp/tsdb-demo
(umask 077 && echo 'try-out-password-17' > /tmp/tsdb-demo/password)
./build/tsdb /tmp/tsdb-demo/db user add admin --superuser --password-file /tmp/tsdb-demo/password
./build/tsdbd --dir /tmp/tsdb-demo/db

It listens on port 7070. Leave it running and open a second terminal.

make has built the client already. Include client/tsdb_client.h (and include/tsdb_calendar.h for time zones) and link with build/libtsdbclient.a -lzstd -lpthread -lm. The C client does not contain the database; it only talks to the server. For the database inside your program, include include/tsdb.h and link build/libtsdb.a instead.

The package is intellistream-tsdb-client; in code it is tsdb_client. Add it to your Cargo.toml:

[dependencies]
intellistream-tsdb-client = { path = "path/to/tsdb/client/rust" }

You need Rust (cargo) and Python 3.9 or newer. Build the module into a virtual environment:

cd client/python
python3 -m venv /tmp/tsdb-venv
/tmp/tsdb-venv/bin/pip install maturin
VIRTUAL_ENV=/tmp/tsdb-venv /tmp/tsdb-venv/bin/maturin develop --release

Then import tsdb_client.

You need Java 17 or newer and Maven. Install the client into your local Maven repository with mvn install in client/java, and add it to your project:

<dependency>
  <groupId>ai.intellistream</groupId>
  <artifactId>tsdb-client</artifactId>
  <version>0.11.0</version>
</dependency>

2. Write a day and read it back

This program connects to tsdbd and gets the id of the series pump7.flow (the first time, that creates it). It writes one reading a minute for a day, then reads it back three ways: the newest readings, a summary per hour, and the sum per day where days begin at midnight in Oslo.

/* tsdb as a server: write one day of readings through tsdbd and read them back.
 *
 *   build/tsdb /tmp/tsdb-qs/server user add admin --superuser --password-file /tmp/tsdb-qs/password
 *   build/tsdbd --dir /tmp/tsdb-qs/server --port 17070 &
 *   build/client-example 127.0.0.1 17070 admin /tmp/tsdb-qs/password
 *
 * The last two arguments are the user and a file whose first line is the password. Leave them out for a
 * tsdbd started with --no-auth. The C client does not link the storage engine; it only talks to the server. */
#include <inttypes.h>
#include <math.h>
#include <stdio.h>
#include <stdlib.h>

#include "tsdb_calendar.h"
#include "tsdb_client.h"

#define N 1440 /* one reading a minute for a day */

/* Called with blocks of rows. We only count them and remember the first one. */
struct seen {
    size_t rows;
    int64_t first_ts;
    float first_value;
};

static int on_rows(void *ctx, const int64_t *ts, const float *values, size_t n)
{
    struct seen *s = ctx;
    if (s->rows == 0 && n > 0) {
        s->first_ts = ts[0];
        s->first_value = values[0];
    }
    s->rows += n;
    return 0; /* non-zero would stop early */
}

#define CHECK(call)                                                         \
    do {                                                                    \
        int err_ = (call);                                                  \
        if (err_ < 0) {                                                     \
            fprintf(stderr, "%s failed: %s\n", #call, tsdbc_strerror(err_)); \
            rc = 1;                                                         \
            goto done;                                                      \
        }                                                                   \
    } while (0)

int main(int argc, char **argv)
{
    const char *host = argc > 1 ? argv[1] : "127.0.0.1";
    int port = argc > 2 ? atoi(argv[2]) : 7070;
    tsdbc_options login = {0};
    login.user = argc > 3 ? argv[3] : NULL;          /* NULL: no login */
    login.password_file = argc > 4 ? argv[4] : NULL; /* never the password itself: argv is visible in ps */
    const int64_t start = 1717200000000; /* 2024-06-01 00:00:00 UTC, in milliseconds */
    const int64_t end = start + 86400000;

    int rc = 0;
    tsdbc *c = NULL;
    tsdb_bucket *hours = NULL, *days = NULL;
    CHECK(tsdbc_connect_opts(host, port, &login, &c)); /* on an error, CHECK prints it and jumps to done: below */

    tsdb_sid flow;
    CHECK(tsdbc_series(c, "pump7.flow", 1, &flow)); /* 1: create the series if it is missing */

    static int64_t times[N];
    static float values[N];
    for (int i = 0; i < N; i++) {
        times[i] = start + (int64_t)i * 60000;
        values[i] = 50.0f + 10.0f * sinf((float)i / 229.0f);
    }
    CHECK(tsdbc_insert_series(c, flow, times, values, N)); /* the server has it on disk when this returns */
    printf("wrote %d rows\n", N);

    struct seen seen = {0};
    CHECK(tsdbc_query_raw(c, flow, start, end, on_rows, &seen));
    printf("read %zu rows, the first: %" PRId64 " %.2f\n", seen.rows, seen.first_ts, seen.first_value);

    size_t nhours;
    CHECK(tsdbc_query_agg(c, flow, start, end, 3600000, &hours, &nhours)); /* one summary per hour */
    if (nhours > 0) /* an empty time range gives no buckets */
        printf("%zu hours, the first average: %.2f\n", nhours, tsdb_bucket_avg(&hours[0]));

    size_t ndays;
    tsdb_group day = {TSDB_DAY, 1, "Europe/Oslo"};
    CHECK(tsdbc_query_group(c, flow, start, end, &day, NULL, &days, &ndays)); /* one summary per Oslo day */
    for (size_t i = 0; i < ndays; i++)
        printf("Oslo day %zu: %" PRIu64 " rows, average %.2f\n", i + 1, days[i].count, tsdb_bucket_avg(&days[i]));

done:
    free(hours);
    free(days);
    if (c)
        tsdbc_close(c);
    return rc;
}

examples/client.c. make builds it; run build/client-example 127.0.0.1 7070 admin /tmp/tsdb-demo/password.

use tsdb_client::{Client, ConnectOptions, Group, QueryOpts, Unit};

fn main() -> Result<(), tsdb_client::Error> {
    // Connect to tsdbd and log in. The arguments, all optional: the address, the user and its password file.
    let mut args = std::env::args().skip(1);
    let addr = args.next().unwrap_or_else(|| "127.0.0.1:7070".to_string());
    let user = args.next().unwrap_or_else(|| "admin".to_string());
    let password_file = args.next().unwrap_or_else(|| "/tmp/tsdb-demo/password".to_string());
    let login = ConnectOptions::new().user(user).password_file(password_file);
    let mut client = Client::connect_opts(addr.as_str(), &login)?;

    // Get the id of a series (it is created the first time).
    let id = client.series("pump7.flow")?;

    // Insert one value per minute for a day. Times are ms since 1970-01-01 UTC.
    let t0: i64 = 1_699_999_200_000; // 2023-11-14 22:00 UTC
    let ts: Vec<i64> = (0..1440).map(|i| t0 + i * 60_000).collect();
    let values: Vec<f32> = (0..1440).map(|i| (i % 60) as f32).collect();
    client.insert_series(id, &ts, &values)?;

    // Read the newest three values.
    for (t, v) in client.query_points(id, .., QueryOpts::desc().limit(3))?.iter() {
        println!("raw  {t} {v}");
    }

    // Count, average, min and max per hour, for the first six hours.
    for b in client.query_agg(id, t0..t0 + 6 * 3_600_000, 3_600_000)? {
        println!("hour {} count {} avg {:.2} min {} max {}", b.start, b.count, b.avg(), b.min, b.max);
    }

    // Sum per day, where days begin at midnight in Oslo.
    let days = Group::new(Unit::Day).timezone("Europe/Oslo");
    for b in client.query_group(id, .., Some(&days), QueryOpts::asc())? {
        println!("day  {} count {} sum {}", b.start, b.count, b.sum);
    }

    client.close()
}

The quick start of client/python/README.md. Save it as quickstart.py and run it with /tmp/tsdb-venv/bin/python quickstart.py.

package ai.intellistream.tsdb.example;

import ai.intellistream.tsdb.Bucket;
import ai.intellistream.tsdb.ConnectOptions;
import ai.intellistream.tsdb.Group;
import ai.intellistream.tsdb.Order;
import ai.intellistream.tsdb.Points;
import ai.intellistream.tsdb.TsdbClient;
import ai.intellistream.tsdb.Unit;

import java.nio.file.Path;
import java.time.Instant;

/**
 * Quick start: {@code java -cp tsdb-client.jar ai.intellistream.tsdb.example.Example [host] [port] [user]
 * [password-file]}.
 */
public final class Example {
    private Example() {
    }

    /**
     * Runs the example.
     *
     * @param args the host and port of a running tsdbd, the user to log in as and the file with its password;
     *     default 127.0.0.1 7070 admin /tmp/tsdb-demo/password
     * @throws Exception if the server cannot be reached, the login fails or a request fails
     */
    public static void main(String[] args) throws Exception {
        String host = args.length > 0 ? args[0] : "127.0.0.1";
        int port = args.length > 1 ? Integer.parseInt(args[1]) : TsdbClient.DEFAULT_PORT;
        String user = args.length > 2 ? args[2] : "admin";
        Path passwordFile = Path.of(args.length > 3 ? args[3] : "/tmp/tsdb-demo/password");

        // Log in as user, with the password from the first line of passwordFile.
        ConnectOptions login = ConnectOptions.defaults().user(user).passwordFile(passwordFile);
        try (TsdbClient client = TsdbClient.connect(host, port, login)) {
            long pump = client.series("pump7.flow_m3h"); // get or create

            // One reading per minute for a day, from 2024-03-01T00:00Z, as a single batch.
            long t0 = Instant.parse("2024-03-01T00:00:00Z").toEpochMilli();
            int n = 24 * 60;
            long[] ts = new long[n];
            float[] values = new float[n];
            for (int i = 0; i < n; i++) {
                ts[i] = t0 + i * 60_000L;
                values[i] = 40 + (i % 60) / 10f;
            }
            client.insert(pump, ts, values);

            // Raw points: the three newest.
            Points newest = client.queryPoints(pump, Long.MIN_VALUE, Long.MAX_VALUE, Order.DESC, 3);
            for (int i = 0; i < newest.size(); i++)
                System.out.printf("%s %.1f%n", Instant.ofEpochMilli(newest.timestamp(i)), newest.value(i));

            // Aggregates over fixed 6-hour intervals.
            for (Bucket b : client.queryAgg(pump, t0, t0 + 86_400_000L, 6 * 3_600_000L))
                System.out.printf("%s count %d avg %.2f min %.1f max %.1f%n", Instant.ofEpochMilli(b.start()),
                        b.count(), b.avg(), b.min(), b.max());

            // Calendar days in Oslo (UTC+1 in March): the UTC day spans two local days.
            Group osloDays = Group.of(Unit.DAY).in("Europe/Oslo");
            for (Bucket b : client.queryGroup(pump, Long.MIN_VALUE, Long.MAX_VALUE, osloDays))
                System.out.printf("Oslo day from %s: %d points%n", Instant.ofEpochMilli(b.start()), b.count());
        }
    }
}

3. The database inside your program

The same work with no server and no network: the database is a directory, opened by your program.

 *

use tsdb_client::{Group, QueryOpts, Unit};
use tsdb_embedded::{Db, Options};

fn main() -> tsdb_client::Result<()> {
    let dir = std::env::args().nth(1).unwrap_or_else(|| "/tmp/tsdb-embedded".into());
    let db = Db::open(&dir, &Options::default())?; // the engine runs in this process

    let s = db.series("pump7.flow")?;
    let t0 = 1_700_000_000_000; // one row a minute for a day
    let ts: Vec<i64> = (0..1440).map(|i| t0 + i * 60_000).collect();

    db.insert_series(s, &ts, &values)?;

    let p = db.query_points(s, t0..t0 + 180_000, QueryOpts::asc())?;
    for (t, v) in p.ts.iter().zip(&p.values) {
        println!("raw  {t} {v}");
    }
    let days = Group::new(Unit::Day).timezone("Europe/Oslo");
    for b in db.query_group(s, .., Some(&days), QueryOpts::asc())? {
        println!("day  {} {} {}", b.start, b.count, b.sum);
    }
    db.close()
}

4. Spread, time-weighted averages and charts

Three questions that a plain average does not answer: how much do the values spread, what was the average over time for a sensor that reports only on change, and which few points draw the same line. Concepts explains each.

The calls in include/tsdb.h. The C client has the same ones with the prefix tsdbc_ (tsdbc_query_spread, tsdbc_query_twa, tsdbc_query_lttb).

                    void *ctx);
/* The same points newest first (TSDB_DESC), or only the first `limit` of them in that order: the choice
 * does not change, since it needs every point of the range anyway (opts NULL: as tsdb_query_lttb). Newest first
 * holds at most min(limit, threshold) chosen points in memory, and none where the threshold keeps every point
 * (a backwards read that stops at the limit): 8 bytes and the value's width each, so a threshold near the rows of
 * a very large range holds about as much as reading the range raw. tsdbd caps the threshold of its clients
 * (--lttb-max-threshold, default 16,777,216). */
int tsdb_query_lttb_ex(tsdb *db, tsdb_sid series, int64_t start, int64_t end, size_t threshold,
                       const tsdb_query_opts *opts, tsdb_values_fn fn, void *ctx);
 *    computed from exact moments, with a few roundings at the end (in long
 *    double), and does not depend on the order of the rows.
 * Like every aggregate, the spread covers only the rows that are left: a
 * deleted, replaced-away or expired (retention) value is never part of it,
 * whether merges have removed it from disk yet or not.
 * The spread costs a second pass over decoded floats and the squares of
 * integers, so only these functions compute it; tsdb_query_agg and the rest
 * do not. Arguments, order and limit as in tsdb_query_agg, tsdb_query_group_ex
 * and tsdb_query_bounds; opts may be NULL. Free with free(). */
typedef struct tsdb_spread_bucket {
    tsdb_bucket b;
double sum;
double min, max;
use tsdb_client::{Client, ConnectOptions, QueryOpts, TwaOpts, ValueType};

fn main() -> Result<(), tsdb_client::Error> {
    // As in the quick start: the address, the user and its password file.
    let mut args = std::env::args().skip(1);
    let addr = args.next().unwrap_or_else(|| "127.0.0.1:7070".to_string());
    let user = args.next().unwrap_or_else(|| "admin".to_string());
    let password_file = args.next().unwrap_or_else(|| "/tmp/tsdb-demo/password".to_string());
    let login = ConnectOptions::new().user(user).password_file(password_file);
    let mut client = Client::connect_opts(addr.as_str(), &login)?;
    let t0: i64 = 1_699_999_200_000; // 2023-11-14 22:00 UTC
    let hour = 3_600_000;

    // A tank level, sent only when it changes: 10 until 22:45, then 70, then 40 at midnight.
    let tank = client.series_create("tank3.level", ValueType::Float64, None)?;
    let at = [t0, t0 + 45 * 60_000, t0 + hour, t0 + 2 * hour];
    client.insert_series_values(tank, &at, &[10.0f64, 70.0, 70.0, 40.0])?;

    // The plain average counts each reading once; the time-weighted one counts how long each value held.
    let plain = client.query_agg(tank, t0..t0 + 2 * hour, hour)?;
    let twa = client.query_twa(tank, t0..t0 + 2 * hour, hour, TwaOpts::step())?;
    for (p, w) in plain.iter().zip(&twa) {
        println!("hour {} avg {:.1} twa {:.1}", p.start, p.avg(), w.twa);
    }

    // How far the readings spread: range and standard deviation of the whole series.
    let s = &client.query_spread(tank, .., 0, QueryOpts::asc())?[0];
    println!("spread range {} stddev {:.2}", s.range, s.stddev_samp);

    // 10,000 vibration readings cut to 100 points that keep the shape of the line, for a chart.
    let vib = client.series("pump7.vibration")?;
    let ts: Vec<i64> = (0..10_000).map(|i| t0 + i * 1000).collect();
    let values: Vec<f32> = (0..10_000).map(|i| (i as f32 / 300.0).sin()).collect();
    client.insert_series(vib, &ts, &values)?;
    let chart = client.query_lttb(vib, .., 100, QueryOpts::asc())?;
    println!("chart {} of {} points", chart.len(), ts.len());

    client.close()
}

    print("spread", b.range, b.var_pop, b.stddev_pop)

    (w,) = db.query_twa(s, 0, 5000)                  # each value weighted by how long it held
    print("twa", w.twa, w.covered_ms)

    ts, values = db.query_lttb(s, 3, numpy=False)    # at most 3 points that keep the shape
    print("chart", ts)

    last = db.query_agg(s, None, None, 1000, order="desc", limit=1)
    print("last", last[0].start, last[0].sum)

5. Many series in one request

One request, one answer per series, in the order of your list, all read at the same instant.

tsdb_query_multi calls your function once per series as the answers come; tsdb_query_multi_collect returns them all at once. The C client has tsdbc_query_multi and tsdbc_query_multi_collect.

*  - trim (with bounds): TSDB_MULTI_TRIM_LO starts series i's bounds at the last one at or before its first row
*    in range, TSDB_MULTI_TRIM_HI ends them at the last one at or before its last row (a series without rows then
*    has no buckets). That is what a client does for local periods, so one union of bounds serves every series.
*    The whole bounds must strictly increase, trimmed or not; if they do not, every series answers TSDB_ERR_INVALID.
* Calls to fn: VALUES and LTTB send rows (ts, values in the API widths, valid as in tsdb_values) in one or more
* calls with last = 0 and then one with n = 0 and last = 1, or all of them in one call with last = 1; the other

The Rust client has the same many-series calls. Its README shows them.

import numpy as np
import tsdb_client

with tsdb_client.Client("127.0.0.1", port=7070, user="admin", password_file="/tmp/tsdb-demo/password") as db:
    t0 = 1_700_000_000_000
    ts = t0 + np.arange(1440, dtype=np.int64) * 60_000  # one row a minute for a day
    pumps = [db.series(f"pump{i}.flow") for i in range(3)]
    for i, s in enumerate(pumps):
        db.insert_series(s, ts, np.full(1440, i + 1.0, dtype=np.float32))

    # hourly buckets of every pump in one call; numpy over all of them at once
    r = db.query_buckets_multi(pumps, t0, t0 + 86_400_000, interval_ms=3_600_000)
    print(len(r), r.offsets.tolist(), r.sum.sum())
    print(len(r[1]), r[1].sum[0])  # pump 1 alone, as query_agg() gives it

    # the newest row of every pump
    last = db.query_values_multi(pumps, order="desc", limit=1)
    print(last.values.tolist())

7. Logging in, users and rights

Every program on this page logs in with a user name and a password. The password never crosses the network: client and server prove to each other that they know it. A wrong user name and a wrong password give the same error. Keep passwords in a file only you can read, or wherever your program keeps secrets, never in the source.

A superuser may do everything. Every other user has rights (read, insert, delete, admin) on the series a selector matches: all, a name prefix such as prefix:acme/, one name, or a pattern such as glob:acme/*/temp. A series a user may not touch answers "permission denied", whether it exists or not. A superuser, or a user with admin on a prefix, adds users and gives them rights over the connection. Here a superuser adds a user for a collector that may only add rows under acme/, which then cannot read them back nor see another tenant's series. On the server's own machine an operator does the same with tsdb DIR user.

The login is part of the connect options:

tsdbc_options login = {0};
login.user = argc > 3 ? argv[3] : NULL;          /* NULL: no login */
login.password_file = argc > 4 ? argv[4] : NULL; /* never the password itself: argv is visible in ps */

and the calls for users are in client/tsdb_client.h (the kinds of selector are TSDB_SEL_*, the rights TSDB_RIGHT_* bits):

void tsdbc_users_free(tsdbc_user *u, size_t n);
int tsdbc_user_create(tsdbc *c, const char *name, const char *password, uint32_t iterations, uint8_t flags);
int tsdbc_user_drop(tsdbc *c, const char *name);
int tsdbc_user_alter(tsdbc *c, const char *name, uint8_t flags);
int tsdbc_user_password(tsdbc *c, const char *name, const char *password, uint32_t iterations);
int tsdbc_grant(tsdbc *c, const char *name, uint8_t kind, const char *selector, uint8_t rights);

use tsdb_client::{Client, ConnectOptions, QueryOpts, Rights, Selector};

fn main() -> Result<(), tsdb_client::Error> {
    // Log in as the superuser of the quick start: the address, the user and its password file.
    let mut args = std::env::args().skip(1);
    let addr = args.next().unwrap_or_else(|| "127.0.0.1:7070".to_string());
    let user = args.next().unwrap_or_else(|| "admin".to_string());
    let password_file = args.next().unwrap_or_else(|| "/tmp/tsdb-demo/password".to_string());
    let mut admin = Client::connect_opts(addr.as_str(), &ConnectOptions::new().user(user).password_file(password_file))?;

    // A user for a collector of the tenant "acme": it may add rows to the series whose names start with
    // "acme/", and do nothing else.
    admin.user_create("acme/ingest", "collector-password-5", false, None)?;
    admin.grant("acme/ingest", &Selector::Prefix("acme/".into()), Rights::INSERT)?;
    for u in admin.user_list()? {
        let grants: Vec<String> = u.grants.iter().map(|g| format!("{} on {}", g.rights, g.selector)).collect();
        println!("{}: {}", u.name, if u.superuser { "superuser".to_string() } else { grants.join(", ") });
    }

    // The collector logs in with its own password and writes a row.
    let login = ConnectOptions::new().user("acme/ingest").password("collector-password-5");
    let mut collector = Client::connect_opts(addr.as_str(), &login)?;
    let pump = collector.series("acme/pump1")?;
    collector.insert_series(pump, &[1_700_000_000_000], &[1.5])?;

    // It may not read the row back, nor see a series of another tenant.
    if let Err(e) = collector.query_points(pump, .., QueryOpts::asc()) {
        println!("read acme/pump1: {e}");
    }
    if let Err(e) = collector.series("globex/pump1") {
        println!("globex/pump1: {e}");
    }
    collector.close()?;

    // Drop the user again, so that the example can run once more.
    admin.user_drop("acme/ingest")?;
    admin.close()
}
